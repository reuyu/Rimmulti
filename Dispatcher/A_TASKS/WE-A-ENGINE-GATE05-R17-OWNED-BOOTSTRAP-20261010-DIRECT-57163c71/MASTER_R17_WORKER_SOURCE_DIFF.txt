+        public static string ComputeFileSha256(string filePath)
+        {
+            if (string.IsNullOrEmpty(filePath) || !File.Exists(filePath))
+            {
+                throw new FileNotFoundException($"File not found for hash calculation: '{filePath}'");
+            }
+            using (var sha = SHA256.Create())
+            using (var stream = File.Open(filePath, FileMode.Open, FileAccess.Read, FileShare.ReadWrite))
+            {
+                byte[] hash = sha.ComputeHash(stream);
+                StringBuilder sb = new StringBuilder(hash.Length * 2);
+                for (int i = 0; i < hash.Length; i++)
+                {
+                    sb.Append(hash[i].ToString("X2"));
+                }
+                return sb.ToString();
+            }
+        }

+                return (false, "Process MainModule FileName is empty or inaccessible.");
+            }
+            string normalizedMain;
+            try
+            {
+                normalizedMain = Path.GetFullPath(mainModulePath.Trim());
+            }
+            catch (Exception ex)
+            {
+                return (false, $"Failed to normalize process MainModule path: {ex.Message}");
+            }
+
+            string normalizedExpected;
+            try
+            {
+                normalizedExpected = Path.GetFullPath(expectedExePath.Trim());
+            }
+            catch (Exception ex)
+            {
+                return (false, $"Failed to normalize expected executable path: {ex.Message}");
+            }
+
+            if (!string.Equals(normalizedMain, normalizedExpected, StringComparison.OrdinalIgnoreCase))
+            {
+                return (false, $"Process MainModule path mismatch. Expected='{normalizedExpected}', Observed='{normalizedMain}'.");
+            }
+
+            var exeCheck = VerifyAdmittedExecutable(normalizedMain, expectedSha256);
+            if (!exeCheck.isValid)
+            {
+                return (false, $"Process MainModule executable verification failed: {exeCheck.error}");
+            }
+
+            return (true, string.Empty);
+        }
+    }
